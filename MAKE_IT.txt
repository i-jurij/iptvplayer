**Вариант 2: wxGLCanvas с собственным render loop для mpv.**

Это **единственный по-настоящему надёжный способ**, потому что:

1. **wxWidgets имеет встроенную поддержку wxGLCanvas** — это не гибридный костыль
2. **Работает и в X11, и в Wayland** — потому что не полагается на встраивание окон
3. **mpv render context API стабилен** — поддерживается с mpv 0.28+, у тебя 0.40.0
4. **Нет зависимости от специфики GTK или Wayland** — всё на уровне OpenGL

## Как это реализовать

### Шаг 1: Базовый wxGLCanvas

```c++
#include <wx/glcanvas.h>
#include <mpv/client.h>
#include <mpv/render_gl.h>

class MpvGLCanvas : public wxGLCanvas {
public:
    MpvGLCanvas(wxWindow* parent);
    ~MpvGLCanvas();
    
    void LoadVideo(const std::string& path);
    
private:
    mpv_handle* mpv_handle = nullptr;
    mpv_render_context* render_ctx = nullptr;
    std::unique_ptr<wxGLContext> gl_context;
    
    void OnPaint(wxPaintEvent& evt);
    void OnSize(wxSizeEvent& evt);
    void RenderFrame();
    
    wxDECLARE_EVENT_TABLE();
};
```

### Шаг 2: Инициализация OpenGL контекста

```c++
MpvGLCanvas::MpvGLCanvas(wxWindow* parent)
    : wxGLCanvas(parent, wxID_ANY, wxGLCanvasAttrs()
        .PlatformDefaults()
        .Defaults()) {
    
    // Создать OpenGL контекст
    gl_context = std::make_unique<wxGLContext>(this);
    
    // Инициализировать mpv
    mpv_handle = mpv_create();
    mpv_set_option_string(mpv_handle, "vo", "null");  // Отключить встроенный вывод
    mpv_set_option_string(mpv_handle, "ao", "null");  // Отключить встроенный звук
    mpv_initialize(mpv_handle);
    
    // Создать render context
    SetCurrent(*gl_context);
    
    mpv_opengl_init_params gl_init_params{
        .get_proc_address = [](void*, const char* name) {
            return (void*)wxGLCanvas::GetProcAddress(name);
        }
    };
    
    int vo_flip_y = 1;
    mpv_render_param params[] = {
        {MPV_RENDER_PARAM_API_TYPE, &(int){MPV_RENDER_API_TYPE_OPENGL}},
        {MPV_RENDER_PARAM_OPENGL_INIT_PARAMS, &gl_init_params},
        {MPV_RENDER_PARAM_FLIP_Y, &vo_flip_y},
        {0}
    };
    
    mpv_render_context_create(&render_ctx, mpv_handle, params);
    
    // Привязать события
    Bind(wxEVT_PAINT, &MpvGLCanvas::OnPaint, this);
    Bind(wxEVT_SIZE, &MpvGLCanvas::OnSize, this);
}
```

### Шаг 3: Рендеринг

```c++
void MpvGLCanvas::OnPaint(wxPaintEvent& evt) {
    wxPaintDC dc(this);
    SetCurrent(*gl_context);
    
    RenderFrame();
    SwapBuffers();
}

void MpvGLCanvas::RenderFrame() {
    // Получить размер окна
    wxSize size = GetClientSize();
    
    // Установить viewport
    glViewport(0, 0, size.GetWidth(), size.GetHeight());
    glClear(GL_COLOR_BUFFER_BIT);
    
    // Рендерить mpv
    if (render_ctx) {
        mpv_render_param params[] = {
            {MPV_RENDER_PARAM_OPENGL_FBO, &(mpv_opengl_fbo){
                .fbo = 0,  // 0 = default framebuffer
                .w = size.GetWidth(),
                .h = size.GetHeight()
            }},
            {0}
        };
        mpv_render_context_render(render_ctx, params);
    }
}

void MpvGLCanvas::OnSize(wxSizeEvent& evt) {
    // Инвалидировать окно для перерисовки
    Refresh();
}

void MpvGLCanvas::LoadVideo(const std::string& path) {
    const char* cmd[] = {"loadfile", path.c_str(), nullptr};
    mpv_command(mpv_handle, cmd);
}
```

### Шаг 4: CMakeLists.txt

```cmake
find_package(wxWidgets COMPONENTS gl REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV mpv)

target_link_libraries(your_app 
    ${wxWidgets_LIBRARIES}
    ${MPV_LIBRARIES}
)

target_include_directories(your_app PRIVATE 
    ${wxWidgets_INCLUDE_DIRS}
    ${MPV_INCLUDE_DIRS}
)
```

## Почему это работает везде

| Компонент | X11 | Wayland | Статус |
|-----------|-----|---------|--------|
| **wxGLCanvas** | ✅ GTK GLX | ✅ GTK EGL | Встроено в wxWidgets |
| **OpenGL** | ✅ GLX | ✅ EGL | Абстрагируется wxWidgets |
| **mpv render context** | ✅ Поддержана | ✅ Поддержана | mpv 0.28+ |
| **Встраивание окон** | ❌ Не требуется | ❌ Не требуется | Используем GL rendering |

**Ключевой момент:** wxGLCanvas сам выбирает GLX (X11) или EGL (Wayland) — тебе это не нужно контролировать. mpv render context тоже платформа-агностичен.


⚠️ Важные нюансы реализации (из вашего анализа):

Инициализация mpv_render_context:


Использовать MPV_RENDER_PARAM_API_TYPE со строкой "opengl" (не &(int){MPV_RENDER_API_TYPE_OPENGL})
mpv_opengl_init_params должен содержать get_proc_address = get_proc_address_impl и get_proc_address_ctx = this
get_proc_address_impl должен быть static и использовать wxGLCanvas::GetProcAddress


Обязательные шаги:


SetCurrent(GetContext()) перед mpv_render_context_create() и mpv_render_context_render()
mpv_set_option_string(mpv, "vo", "null") — отключить встроенный видео вывод
mpv_set_option_string(mpv, "ao", "pulse") — для звукового вывода
FLIP_Y = 1 в mpv_render_param для корректного отображения


Paint loop:


Создать FBO и пересоздавать при изменении размеров
glBindFramebuffer(GL_FRAMEBUFFER, fbo) перед mpv_render_context_render()
glBindFramebuffer(GL_FRAMEBUFFER, 0) после рендеринга


Размеры:


Обновлять FBO размеры через mpv_render_context_render(), а не mpv_command("resize")
OnSize просто вызывает Refresh() для перерисовки


Платформо-специфичные моменты:


Windows: явно задавать --hwdec=auto или --hwdec=cuda
Drag&Drop: переопределить AcceptDropFiles() в MpvGLCanvas
