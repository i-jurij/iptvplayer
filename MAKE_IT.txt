bool EPGDatabase::GetAllMappings(const std::string &playlistId,
                                  std::unordered_map<std::string, MappingEntry> &out) {
    std::lock_guard<std::recursive_mutex> lock(m_mutex);
    if (!m_isOpen) return false;

    try {
        Statement stmt = m_db.PrepareStatement(
            "SELECT key, channel_id, is_manual, ignored FROM playlist_mappings WHERE playlist_id = ?");
        StatementGuard guard(stmt);
        stmt.Bind(1, wxString::FromUTF8(playlistId));
        ResultSet rs = stmt.ExecuteQuery();
        while (rs.NextRow()) {
            std::string key = wx_to_std(rs.GetString(0));
            MappingEntry entry;
            entry.epgId = wx_to_std(rs.GetString(1));
            entry.isManual = rs.GetInt(2) != 0;
            entry.ignored = rs.GetInt(3) != 0;
            out[key] = std::move(entry);
        }
        return true;
    } catch (wxSQLite3::Exception &e) {
        LOG_ERROR("EPGDatabase::GetAllMappings exception: %s", e.GetMessage().ToUTF8().data());
        return false;
    }
}
```

void ManualMappingDialog::ApplySortToMappings() {
    if (!m_mappingList || m_mappingList->GetItemCount() == 0) return;
    m_mappingList->SortItems(
        [this](long item1, long item2, long sortData) -> int {
            int col = m_sortColumn;
            wxString text1 = m_mappingList->GetItemText(item1, col);
            wxString text2 = m_mappingList->GetItemText(item2, col);
            int cmp = wxStricmp(text1, text2);
            if (!m_sortAscending) cmp = -cmp;
            return cmp;
        }, 0);
}

void ManualMappingDialog::OnColumnClick(wxListEvent &event) {
    int col = event.GetColumn();
    if (col == m_sortColumn) {
        m_sortAscending = !m_sortAscending;
    } else {
        m_sortColumn = col;
        m_sortAscending = true;
    }
    ApplySortToMappings();
}
```

### 7.4. Реализация методов фильтрации

**Добавить** в `ManualMappingDialog.cpp`:

```cpp
void ManualMappingDialog::ApplyFilterAndSortPlaylist() {
    m_playlistList->Freeze();
    m_playlistList->DeleteAllItems();
    wxString filter = m_searchPlaylist->GetValue().Lower();
    long idx = 0;
    for (const auto &ch : m_allPlaylistChannels) {
        wxString name = wxString::FromUTF8(ch.getName());
        if (!filter.IsEmpty() && !name.Lower().Contains(filter)) continue;
        wxString tvgId = wxString::FromUTF8(ch.getTvgId());
        long item = m_playlistList->InsertItem(idx, name);
        m_playlistList->SetItem(item, 1, tvgId);
        m_playlistList->SetItemData(item, idx);
        ++idx;
    }
    m_playlistList->Thaw();
}

void ManualMappingDialog::ApplyFilterAndSortEpg() {
    m_epgList->Freeze();
    m_epgList->DeleteAllItems();
    wxString filter = m_searchEpg->GetValue().Lower();
    long idx = 0;
    for (const auto &pair : m_allEpgChannels) {
        wxString name = wxString::FromUTF8(pair.second);
        if (!filter.IsEmpty() && !name.Lower().Contains(filter)) continue;
        wxString id = wxString::FromUTF8(pair.first);
        long item = m_epgList->InsertItem(idx, id);
        m_epgList->SetItem(item, 1, name);
        m_epgList->SetItemData(item, idx);
        ++idx;
    }
    m_epgList->Thaw();
}

void ManualMappingDialog::ApplyFilterAndSortMappings() {
    m_mappingList->Freeze();
    m_mappingList->DeleteAllItems();
    wxString filter = m_searchMappings->GetValue().Lower();
    long idx = 0;
    for (const auto &row : m_allMappingRows) {
        if (!filter.IsEmpty() && !row.channelName.Lower().Contains(filter)) continue;
        long item = m_mappingList->InsertItem(idx, row.tvgId);
        m_mappingList->SetItem(item, 1, row.channelName);
        m_mappingList->SetItem(item, 2, row.epgId);
        m_mappingList->SetItem(item, 3, row.epgName);
        m_mappingList->SetItem(item, 4, row.status);
        m_mappingList->SetItem(item, 5, row.key);
        m_mappingList->SetItemData(item, idx);
        ++idx;
    }
    m_mappingList->Thaw();
    ApplySortToMappings();
}

void ManualMappingDialog::OnSearchPlaylist(wxCommandEvent &) {
    ApplyFilterAndSortPlaylist();
}

void ManualMappingDialog::OnSearchEpg(wxCommandEvent &) {
    ApplyFilterAndSortEpg();
}

void ManualMappingDialog::OnSearchMappings(wxCommandEvent &) {
    ApplyFilterAndSortMappings();
}
```

### 7.5. Заполнение хранилищ в методах заполнения

**В `PopulatePlaylistChannels()`** после получения `channels` сохранить в `m_allPlaylistChannels` и вызвать `ApplyFilterAndSortPlaylist()` вместо прямой вставки.

**В `PopulateEpgChannels()`** аналогично.

**В `PopulateMappings()`** после получения данных заполнить `m_allMappingRows` (вместо прямой вставки), затем вызвать `ApplyFilterAndSortMappings()`.

Пример для `PopulatePlaylistChannels` (полностью):

```cpp
void ManualMappingDialog::PopulatePlaylistChannels() {
    if (!m_mainFrame) return;
    m_allPlaylistChannels = m_mainFrame->GetCurrentChannels();
    // Сортируем полный список
    std::sort(m_allPlaylistChannels.begin(), m_allPlaylistChannels.end(),
              [](const Channel &a, const Channel &b) {
                  return wxStricmp(wxString::FromUTF8(a.getName()),
                                   wxString::FromUTF8(b.getName())) < 0;
              });
    ApplyFilterAndSortPlaylist();
}
```

Аналогично для EPG и маппингов. Для маппингов в `PopulateMappings` заполняем `m_allMappingRows` вектором структур, затем вызываем `ApplyFilterAndSortMappings()`.

========================================
Логические моменты
SetManualMapping: при установке ручного маппинга ты сначала удаляешь любую существующую запись 
(DeleteMappingEntry), а потом вставляешь новую. Это правильно, но стоит помнить: если в БД уже 
был ручной маппинг для другого epgId, он тоже будет удалён. Логически корректно, но может быть 
неожиданно для пользователя.

RemoveChannelMapping: удаляет только ручной маппинг по tvgId. Если ключ был нормализован как "name:...", 
он не удалится этим методом. Это может привести к ситуации, когда ручной маппинг по имени останется висеть.

LoadMappingForPlaylist: проверка хэша плейлиста и количества каналов защищает от несоответствия, 
но если плейлист изменился минимально (например, порядок каналов), маппинг будет признан недействительным. 
Это логически правильно, но может быть слишком строгим.
