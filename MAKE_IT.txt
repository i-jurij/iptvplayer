### Шаг 9. Реализовать новую логику автообновления в `OnAutoUpdateTimer`
```cpp
void EPGManager::OnAutoUpdateTimer(wxTimerEvent&) {
    std::vector<EpgSource> sourcesCopy;
    {
        std::lock_guard<std::recursive_mutex> lock(m_mutex);
        sourcesCopy = m_sources;
    }
    std::vector<std::string> urlsToUpdate;
    for (const auto& src : sourcesCopy) {
        if (src.autoUpdate) urlsToUpdate.push_back(src.url);
    }
    if (urlsToUpdate.empty()) return;

    // Счётчик завершённых
    std::atomic<int> completed{0};
    std::atomic<bool> anySuccess{false};
    for (const auto& url : urlsToUpdate) {
        // Найти имя источника
        std::string name;
        for (const auto& src : sourcesCopy) {
            if (src.url == url) { name = src.name; break; }
        }
        RefreshSourceAsync(url, name, [this, &completed, &anySuccess, urlsToUpdate](bool success, const std::string&) {
            if (success) anySuccess = true;
            if (++completed == (int)urlsToUpdate.size()) {
                // Все обновления завершены
                if (anySuccess && !m_currentPlaylistId.empty() && m_playlistManager) {
                    Playlist* pl = m_playlistManager->findByUniqueId(m_currentPlaylistId);
                    if (pl && !pl->getChannels().empty()) {
                        MatchChannelsAsync(pl->getChannels(), m_currentPlaylistId, nullptr);
                    }
                }
            }
        });
    }
}
```
